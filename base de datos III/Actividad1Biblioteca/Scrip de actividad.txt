CREACION DE TABLA
---------------------------------
CREATE TABLE libros(
id_libro NUMBER(5) PRIMARY KEY,
titulo varchar2(60) NOT NULL,
autor varchar2(60),
fecha_publicacion DATE,
disponible number(1)
);

SELECT * FROM libros

------------------------------------------
MOSTRAR POR TITULO Y AUTOR Y ORDENADOS ALFABETICAMENTE

SELECT titulo, autor FROM libros
ORDER BY titulo ASC, autor ASC;
-------------------------------------------
ACTUALIZAR CAMPO DE FILA POR MEDIO DE ID
